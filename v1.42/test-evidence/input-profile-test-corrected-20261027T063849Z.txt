import re
from pathlib import Path

path = Path("examples/test-vectors.md")
raw = path.read_bytes()
text = raw.decode("utf-8")
lines = text.splitlines()

# Extract the two documented vector tables.
wanted = {"V-IN-", "V-PRO-"}
rows = []
for line in lines:
    if not line.lstrip().startswith("|"):
        continue
    cells = [c.strip() for c in line.strip().strip("|").split("|")]
    if not cells:
        continue
    ident = cells[0]
    if any(ident.startswith(prefix) for prefix in wanted):
        rows.append((ident, cells))

input_rows = [(i, c) for i, c in rows if i.startswith("V-IN-")]
profile_rows = [(i, c) for i, c in rows if i.startswith("V-PRO-")]

checks = []
checks.append(("source decodes as UTF-8", True))
checks.append(("10 input-format vectors found", len(input_rows) == 10))
checks.append(("5 profile vectors found", len(profile_rows) == 5))

# Verify table shape and expected error-code references, not runtime behavior.
input_expected = {
    "V-IN-01": "S64-E011", "V-IN-02": "S64-E011",
    "V-IN-03": "S64-E011", "V-IN-04": "S64-E011",
    "V-IN-05": "S64-E011", "V-IN-06": "S64-E011",
    "V-IN-06b": "S64-E011", "V-IN-07": "S64-E001",
    "V-IN-08": "S64-E001", "V-IN-09": "S64-E001",
}
profile_expected = {f"V-PRO-{n:02d}": "S64-E002" for n in range(1, 6)}

def validate(table_rows, expected, label):
    seen = {}
    for ident, cells in table_rows:
        if len(cells) < 3:
            continue
        outcome = next(
            (cell for cell in cells[2:] if re.search(r"S64-E\d{3}", cell)),
            None
        )
        if outcome:
            match = re.search(r"S64-E\d{3}", outcome)
            seen[ident] = match.group(0)
        print(f"{label} ROW: {ident} | {' | '.join(cells[1:])}")
    return seen

input_seen = validate(input_rows, input_expected, "INPUT")
profile_seen = validate(profile_rows, profile_expected, "PROFILE")

checks.append((
    "input vector expected codes match documented table",
    all(input_seen.get(k) == v for k, v in input_expected.items())
    and set(input_seen) == set(input_expected)
))
checks.append((
    "profile vector expected codes match documented table",
    all(profile_seen.get(k) == v for k, v in profile_expected.items())
    and set(profile_seen) == set(profile_expected)
))

for name, ok in checks:
    print(f"{'PASS' if ok else 'FAIL'}: {name}")

failed = sum(not ok for _, ok in checks)
print(f"RESULT: {len(checks)-failed}/{len(checks)} checks passed")
print("SCOPE: documentation table consistency only; no implementation executed")
raise SystemExit(1 if failed else 0)

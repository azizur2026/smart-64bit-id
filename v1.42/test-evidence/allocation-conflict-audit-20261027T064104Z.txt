import re
from pathlib import Path

    Path("examples/test-vectors.md"),

]

terms = re.compile(
    r"allocation_conflict|S64-E009|E009|E030|"
    r"canonical allocation|lifecycle State|"
    r"same Engine|same.*Local|retirement|history-chain",
    re.IGNORECASE,
)

checks = []
all_text = {}

for path in files:
    if not path.is_file():
        print(f"FILE MISSING: {path}")
        checks.append((f"{path} exists", False))
        continue

    raw = path.read_bytes()
    text = raw.decode("utf-8")
    all_text[path.as_posix()] = text
    print(f"FILE: {path}")
    print(f"BYTES: {len(raw)}")
    print(f"SHA256: {hashlib.sha256(raw).hexdigest()}")

    matches = [
        (i, line.strip())
        for i, line in enumerate(text.splitlines(), 1)
        if terms.search(line)
    ]
    print(f"MATCHING_LINES: {len(matches)}")
    for number, line in matches:
        print(f"  L{number}: {line}")

    checks.append((f"{path} decodes as UTF-8", True))
    checks.append((f"{path} contains relevant allocation references",
                   bool(matches)))

# Verify that the key policy language is present across the documents.
combined = "\n".join(all_text.values()).lower()

checks.append((
    "E009 appears in the reviewed documents",
    "e009" in combined
))
checks.append((
    "E030 is not defined as a current v1.42 error code in these documents",
    not re.search(r"\bS64-E030\b", combined)
))
checks.append((
    "duplicate canonical allocation policy is explicitly mentioned",
    "canonical allocation" in combined
))
checks.append((
    "lifecycle-state independence is explicitly mentioned",
    "lifecycle state" in combined
))
checks.append((
    "history-chain or retirement handling is mentioned",
    "history-chain" in combined or "retirement" in combined
))

for name, ok in checks:
    print(f"{'PASS' if ok else 'FAIL'}: {name}")

failed = sum(not ok for _, ok in checks)
print(f"RESULT: {len(checks)-failed}/{len(checks)} checks passed")
print("SCOPE: keyword/presence audit only; semantic agreement requires manual review.")
raise SystemExit(1 if failed else 0)

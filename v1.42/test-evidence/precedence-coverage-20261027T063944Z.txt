import hashlib
import re
from pathlib import Path

path = Path("examples/test-vectors.md")
raw = path.read_bytes()
text = raw.decode("utf-8")

# Extract documented vector IDs and their expected error codes.
rows = {}
for line in text.splitlines():
    if not line.lstrip().startswith("|"):
        continue
    cells = [c.strip() for c in line.strip().strip("|").split("|")]
    if not cells:
        continue
    ident = cells[0]
    if re.fullmatch(r"V-(?:IN|PRO)-[0-9]+b?", ident):
        code = next(
            (m.group(0) for cell in cells[1:]
             if (m := re.search(r"S64-E\d{3}", cell))),
            None
        )
        rows[ident] = (cells, code)

checks = []

))

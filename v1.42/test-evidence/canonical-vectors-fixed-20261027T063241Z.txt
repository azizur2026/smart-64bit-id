import re
from pathlib import Path

p = Path("examples/test-vectors.md")
raw = p.read_bytes()
text = raw.decode("utf-8")
lines = text.splitlines()

rows = []
for line in lines:
    if not re.search(r"\|\s*V-CLASS-", line):
        continue
    cells = [c.strip() for c in line.strip().strip("|").split("|")]
    if len(cells) < 7:
        continue
    ident = re.fullmatch(r"V-CLASS-([01]{4})", cells[0])
    hx = re.fullmatch(r"`?([0-9A-Fa-f]{16})`?", cells[1])
    dec = re.fullmatch(r"[\d,]+", cells[2])
    class_match = re.match(r"([01]{4})\b", cells[3])
    engine = re.fullmatch(r"\d+", cells[4])
    local = re.fullmatch(r"\d+", cells[5])
    role = re.fullmatch(r"[01]", cells[6])
    if all((ident, hx, dec, class_match, engine, local, role)):
        rows.append((
            ident.group(1), hx.group(1), dec.group(0),
            class_match.group(1), int(engine.group()),
            int(local.group()), int(role.group())
        ))

checks = [
    ("source is valid UTF-8", True),
    ("exactly 8 canonical rows found", len(rows) == 8),
]

print(f"Canonical rows parsed: {len(rows)}")
if len(rows) != 8:
    print("Diagnostic: lines containing V-CLASS:")
    for line in lines:
        if "V-CLASS-" in line:
            print(repr(line))
else:
    decimal_ok = fields_ok = profile_ok = width_ok = True
    for bits, hx, dec, class_col, engine, local, role in rows:
        n = int(hx, 16)
        actual = {
            "Class": (n >> 50) & 0xF,
            "Engine": (n >> 29) & 0x1FFFFF,
            "Local": n & 0x1FFFFFFF,
            "Role": (n >> 58) & 1,
        }
        expected = {
            "Class": int(bits, 2),
            "Engine": engine,
            "Local": local,
            "Role": role,
        }
        decimal_ok &= n == int(dec.replace(",", ""))
        fields_ok &= actual == expected and bits == class_col
        profile_ok &= (
            ((n >> 63) & 1) == 0
            and ((n >> 59) & 0xF) == 2
            and ((n >> 54) & 0xF) == 0
        )
        width_ok &= len(bytes.fromhex(hx)) == 8
        print(
            f"VECTOR {bits}: hex={hx} decimal={int(dec.replace(',', ''))} "
            f"actual={actual} expected={expected}"
        )
    checks += [
        ("hexadecimal equals documented decimal", decimal_ok),
        ("Class/Engine/Local/Role fields match table", fields_ok),
        ("Version/Revision/Reserved profile matches", profile_ok),
        ("each ID is exactly 8 bytes", width_ok),
    ]

for label, ok in checks:
    print(f"{'PASS' if ok else 'FAIL'}: {label}")
failed = sum(not ok for _, ok in checks)
print(f"RESULT: {len(checks)-failed}/{len(checks)} checks passed")
raise SystemExit(1 if failed else 0)
